# GP4 V2 300k Robot Readiness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Upgrade the factory from a pilot artifact workflow to a v2, 300k accepted-row, audit-gated workflow that is ready to produce a robot-safe adapter package and a local-install readiness manifest without installing it locally.

**Architecture:** Keep the model as a Semantic IR drafting component only; robot execution remains behind `llm_gateway`, validation, safety, planning, and hardware gates. Add v2 taxonomy, distribution gates, old+new dataset merging, 300k target enforcement, cloud evidence audits, portability cleanup, dependency readiness, and local-install manifests. Use deterministic Python gates for all safety and acceptance decisions; no LLM judgment can pass a gate by itself.

**Tech Stack:** Python 3, pytest, jsonschema, PyYAML, existing cloud-only storage policy, Colab/Kaggle notebooks, Unsloth/Transformers training runtime declared for cloud readiness only.

---

## Review Findings To Address

- Hardcoded paths reduce portability: `Makefile` and config currently assume local paths such as `/home/hieu2/gp4_ws`.
- The pilot adapter is not production-ready: prior metrics were below acceptance thresholds, and local adapter presence must not imply robot readiness.
- Seed gate must remain strict: handwritten seed rows must pass before synthetic generation; v2 should prefer a larger seed base for new risk categories.
- Runtime dependencies are underdeclared for actual cloud train/infer and local adapter loading readiness.
- Cloud workflow currently depends on manual notebook setup and browser launching; v2 should make this explicit and provider-neutral.
- Robot readiness requires evidence: accepted cloud adapter, passed gates, v2 locked eval, package metadata, checksums, and a local-install manifest. It must not install anything in this phase.
- Structured perception feedback is valid for this repo: the dataset should use text commands plus structured `robot_state` and `scene_state` JSON, not image-text pairs.
- The feedback example's `primitive_type` and `execute_motion()` target conflict with the current safety contract. In this factory, the LLM may draft Semantic IR or safe errors only; deterministic adapters and robot execution remain downstream.

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import os
from pathlib import Path
from typing import Mapping


def resolve_gp4_ws(
    *,
    cli_value: str | None,
    env: Mapping[str, str] = os.environ,
) -> Path:
    raw_value = (cli_value or env.get("GP4_WS") or "").strip()
    if not raw_value:
        raise ValueError("GP4_WS is required; set the environment variable or pass --gp4-ws")
    return Path(raw_value).expanduser().resolve(strict=False)


def main() -> int:
    parser = argparse.ArgumentParser(description="Resolve configured GP4 source workspace.")
    parser.add_argument("--gp4-ws")
    parser.add_argument("--print", action="store_true")
    args = parser.parse_args()
    try:
        path = resolve_gp4_ws(cli_value=args.gp4_ws)
    except ValueError as exc:
        print(str(exc))
        return 1
    if args.print:
        print(path)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
from __future__ import annotations

from collections import Counter
from typing import Any

REQUIRED_SCENARIO_TAGS = {
    "singularity",
    "wrist_flip",
    "joint_wrap",
    "timeout_abort_recovery",
    "approval_required",
    "collision_limit_edge",
    "dangerous_os_command",
    "unsupported_tool_hallucination",
}

OPTIONAL_SCENARIO_TAGS = {
    "typo_noise",
    "alias_canonicalization",
    "vision_uncertain",
    "normal_motion",
    "status_query",
}

ALLOWED_SCENARIO_TAGS = REQUIRED_SCENARIO_TAGS | OPTIONAL_SCENARIO_TAGS


def row_scenario_tags(row: dict[str, Any]) -> list[str]:
    metadata = row.get("metadata", {})
    if not isinstance(metadata, dict):
        return []
    raw_tags = metadata.get("scenario_tags", [])
    if raw_tags is None:
        return []
    if not isinstance(raw_tags, list):
        return ["<non-list-scenario-tags>"]
    return [str(tag) for tag in raw_tags]


def invalid_scenario_tags(row: dict[str, Any]) -> list[str]:
    return [tag for tag in row_scenario_tags(row) if tag not in ALLOWED_SCENARIO_TAGS]


def distribution_summary(rows: list[dict[str, Any]]) -> dict[str, Any]:
    tag_counts: Counter[str] = Counter()
    legacy_rows = 0
    for row in rows:
        tags = row_scenario_tags(row)
        if not tags:
            legacy_rows += 1
        tag_counts.update(tags)
    return {
        "rows": len(rows),
        "legacy_rows_without_scenario_tags": legacy_rows,
        "scenario_tags": dict(sorted(tag_counts.items())),
    }


def quota_failures(rows: list[dict[str, Any]], gates: dict[str, Any]) -> list[dict[str, Any]]:
    summary = distribution_summary(rows)
    counts = summary["scenario_tags"]
    minimums = gates.get("scenario_tag_min_counts", {})
    failures: list[dict[str, Any]] = []
    for tag, minimum in sorted(minimums.items()):
        actual = int(counts.get(tag, 0))
        required = int(minimum)
        if actual < required:
            failures.append(
                {
                    "tag": tag,
                    "actual": actual,
                    "minimum": required,
                    "message": f"scenario tag {tag} has {actual} rows but requires {required}",
                }
            )
    return failures
```

```python
from __future__ import annotations

import json
from typing import Any

from factory_common import SEMANTIC_IR_SYSTEM_PROMPT

IMAGE_TEXT_KEYS = {"image", "image_url", "image_base64", "pixels", "rgb_frame", "depth_frame"}
DOWNSTREAM_TRACE_STEPS = {"execute_motion", "send_motoros2", "dispatch_trajectory"}


def structured_context_issues(row: dict[str, Any]) -> list[str]:
    issues: list[str] = []
    scene_state = row.get("scene_state", {})
    if isinstance(scene_state, dict):
        for key in sorted(IMAGE_TEXT_KEYS):
            if key in scene_state:
                issues.append(f"scene_state must contain structured facts only, not {key}")
                return issues
    target_output = row.get("target_output", {})
    if isinstance(target_output, dict) and "primitive_type" in target_output:
        issues.append("target_output must be Semantic IR or safe error; primitive_type belongs downstream")
        return issues
    trace = row.get("react_trace", [])
    if isinstance(trace, list):
        for step in trace:
            if not isinstance(step, dict):
                continue
            name = str(step.get("step", ""))
            if name in DOWNSTREAM_TRACE_STEPS:
                issues.append(
                    f"react_trace step {name} is downstream executor behavior, not LLM training output"
                )
                return issues
    return issues


def build_messages_from_structured_row(row: dict[str, Any]) -> list[dict[str, str]]:
    issues = structured_context_issues(row)
    if issues:
        raise ValueError(issues[0])
```

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
from pathlib import Path
from typing import Any

from cloud_runtime import CloudPathError, validate_cloud_run_paths
from dataset_keys import dataset_identity_key
from factory_common import read_jsonl, write_json, write_jsonl
from v2_taxonomy import distribution_summary


def main() -> int:
    parser = argparse.ArgumentParser(description="Merge old and new accepted GP4 rows.")
    parser.add_argument("--old", action="append", default=[])
    parser.add_argument("--new", action="append", default=[])
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--report", type=Path, required=True)
    parser.add_argument("--target-rows", type=int, required=True)
    parser.add_argument("--cloud-root", type=Path)
    parser.add_argument("--allow-tmp", action="store_true")
    args = parser.parse_args()

    input_paths = [Path(path) for path in [*args.old, *args.new]]
    try:
        validate_cloud_run_paths(
            cloud_root=args.cloud_root,
            dry_run=False,
            inputs=input_paths,
            outputs=[args.output, args.report],
            allow_tmp=args.allow_tmp,
        )
    except CloudPathError as exc:
        print(f"merge_blocked reason={exc} report={args.report}")
        return 1

    old_rows = _read_many([Path(path) for path in args.old])
    new_rows = _read_many([Path(path) for path in args.new])
    selected_rows, dropped_duplicates = select_rows(old_rows, new_rows, args.target_rows)
    passed = len(selected_rows) == args.target_rows
    payload = {
        "passed": passed,
        "blocked_reason": "" if passed else "not enough unique accepted rows",
        "target_rows": args.target_rows,
        "output_rows": len(selected_rows),
        "old_rows_input": len(old_rows),
        "new_rows_input": len(new_rows),
        "old_rows_kept": sum(1 for row in selected_rows if _source_dataset(row) == "old"),
        "new_rows_kept": sum(1 for row in selected_rows if _source_dataset(row) != "old"),
        "dropped_duplicates": dropped_duplicates,
        "distribution": distribution_summary(selected_rows),
    }
    write_json(args.report, payload)
    if passed:
        write_jsonl(args.output, selected_rows)
    print(f"passed={passed} output_rows={len(selected_rows)} report={args.report}")
    return 0 if passed else 1


def select_rows(
    old_rows: list[dict[str, Any]],
    new_rows: list[dict[str, Any]],
    target_rows: int,
) -> tuple[list[dict[str, Any]], int]:
    selected: list[dict[str, Any]] = []
    seen: set[str] = set()
    dropped_duplicates = 0
    for row in [*old_rows, *new_rows]:
        key = dataset_identity_key(row)
        if key in seen:
            dropped_duplicates += 1
            continue
        if len(selected) >= target_rows:
            break
        seen.add(key)
        selected.append(row)
    return selected, dropped_duplicates


def _read_many(paths: list[Path]) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    for path in paths:
        rows.extend(read_jsonl(path))
    return rows


def _source_dataset(row: dict[str, Any]) -> str:
    metadata = row.get("metadata", {})
    if not isinstance(metadata, dict):
        return ""
    return str(metadata.get("source_dataset", ""))


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
#!/usr/bin/env python3
from __future__ import annotations

import argparse
import hashlib
from pathlib import Path
from typing import Any

from check_cloud_storage_policy import CloudStoragePolicy, is_allowed_cloud_path
from cloud_runtime import CloudPathError, validate_cloud_run_paths
from factory_common import read_json, write_json
from package_adapter import adapter_artifact_exists


def main() -> int:
    parser = argparse.ArgumentParser(description="Build local install readiness manifest without installing.")
    parser.add_argument("--adapter-dir", type=Path, required=True)
    parser.add_argument("--acceptance-report", type=Path, required=True)
    parser.add_argument("--target-repo", type=Path, required=True)
    parser.add_argument("--cloud-root", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--allow-tmp", action="store_true")
    args = parser.parse_args()
    try:
        validate_cloud_run_paths(
            cloud_root=args.cloud_root,
            dry_run=False,
            inputs=[args.adapter_dir, args.acceptance_report],
            outputs=[args.output],
            allow_tmp=args.allow_tmp,
        )
    except CloudPathError as exc:
        print(f"local_install_manifest_blocked reason={exc} output={args.output}")
        return 1
    policy = CloudStoragePolicy((args.cloud_root,), allow_tmp=args.allow_tmp)
    manifest = build_manifest(
        adapter_dir=args.adapter_dir,
        acceptance_report=args.acceptance_report,
        target_repo=args.target_repo,
        policy=policy,
    )
    write_json(args.output, manifest)
    print(f"ready_for_local_install={manifest['ready_for_local_install']} output={args.output}")
    return 0 if manifest["ready_for_local_install"] else 1


def build_manifest(
    *,
    adapter_dir: Path,
    acceptance_report: Path,
    target_repo: Path,
    policy: CloudStoragePolicy,
) -> dict[str, Any]:
    acceptance = read_json(acceptance_report)
    blocked_reason = ""
    if not acceptance.get("passed"):
        blocked_reason = "acceptance gate has not passed"
    elif not is_allowed_cloud_path(adapter_dir, policy):
        blocked_reason = "adapter path is not allowed by cloud storage policy"
    elif not adapter_artifact_exists(adapter_dir):
        blocked_reason = "adapter artifact files are missing"

    return {
        "ready_for_local_install": blocked_reason == "",
        "blocked_reason": blocked_reason,
        "target_repo": str(target_repo),
        "acceptance_report": str(acceptance_report),
        "adapter": {
            "path": str(adapter_dir),
            "files": _file_manifest(adapter_dir) if adapter_dir.exists() else {},
        },
        "install_action_performed": False,
        "safety_boundary": "adapter may draft Semantic IR only; robot execution remains behind validation, safety, planning, approval, and hardware gates",
    }


def _file_manifest(adapter_dir: Path) -> dict[str, dict[str, Any]]:
    files: dict[str, dict[str, Any]] = {}
    for path in sorted(adapter_dir.iterdir()):
        if not path.is_file():
            continue
        digest = hashlib.sha256(path.read_bytes()).hexdigest()
        files[path.name] = {"bytes": path.stat().st_size, "sha256": digest}
```
